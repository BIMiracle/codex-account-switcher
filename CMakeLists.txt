cmake_minimum_required(VERSION 3.20)
project(CodexAccountSwitcher VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows + MSVC required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(CodexAccountSwitcher WIN32 src/main.cpp src/core.cpp src/desktop.cpp resources/app.rc)
target_compile_definitions(CodexAccountSwitcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(CodexAccountSwitcher PRIVATE /W4 /utf-8 /permissive- $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
target_link_options(CodexAccountSwitcher PRIVATE /MANIFEST:NO $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
target_link_libraries(CodexAccountSwitcher PRIVATE comctl32 comdlg32 shell32 advapi32 crypt32 ole32 uuid)
enable_testing()
add_test(NAME isolated_self_test COMMAND CodexAccountSwitcher --self-test)
